    p7 = (if hs7 : msr (a1 (a1 (a2 (a2 u)))) (a1 v) < msr u v then op (a1 (a1 (a2 (a2 u)))) (a1 v) else J u v) ∧

    p9 = (if hs9 : msr (p8) (u) < msr u v then op (p8) (u) else J u v) ∧
    op u v = (

  else J u v
    ) :=
  ⟨_, _, _, _, _, _, _, _, _, rfl, rfl, rfl, rfl, rfl, rfl, rfl, rfl, rfl, op.eq_1 u v⟩
